// Question-1

-- a)
SELECT ACC_NO, ACC_TYPE, BALANCE
FROM ACCOUNT
WHERE BRANCH_CODE IN ('B001','B002','B003','B005');

-- b)
SELECT *
FROM CUSTOMER
WHERE CUST_NAME LIKE 'A%' AND ADDRESS = 'Delhi';

-- c)
SELECT CUST_NO, ACC_NO, BALANCE AS ORIGINAL_BALANCE,
       BALANCE * 0.05 AS INTEREST,
       BALANCE + (BALANCE * 0.05) AS FINAL_BALANCE
FROM ACCOUNT;

-- d)
SELECT *
FROM ACCOUNT
WHERE BALANCE BETWEEN 20000 AND 50000;

-- e)
SELECT *
FROM CUSTOMER
WHERE CUST_NAME LIKE '_m%';

-- f)
SELECT ACC_NO
FROM ACCOUNT
WHERE ACC_TYPE = 'Savings'
ORDER BY BALANCE DESC;

-- g)
UPDATE ACCOUNT
SET BALANCE = BALANCE * 1.10;

-- h)
UPDATE ACCOUNT
SET BALANCE = BALANCE * 1.05
WHERE ACC_TYPE = 'Savings' AND BALANCE < 25000;

-- i)
SELECT ACC_NO, BALANCE * 0.06 AS ANNUAL_INTEREST
FROM ACCOUNT;

-- j)
UPDATE ACCOUNT
SET ACC_TYPE = 'Savings'
WHERE ACC_TYPE = 'Current' AND BALANCE < 45000;

-- k)
DELETE FROM CUSTOMER
WHERE CUST_NO IN (
    SELECT CUST_NO FROM LOAN WHERE AMOUNT < 150000
);

-- l)
DELETE FROM CUSTOMER
WHERE CUST_NAME LIKE '%h';

// Question-2

-- a)
SELECT AVG(BALANCE) AS AVG_SAVINGS_BALANCE
FROM ACCOUNT
WHERE ACC_TYPE = 'Savings';

-- b)
SELECT TRANS_TYPE, AVG(TRANS_AMOUNT) AS AVG_AMOUNT
FROM TRANSACTION_DETAILS
GROUP BY TRANS_TYPE;

-- c)
SELECT ACC_NO, CUST_NO, BALANCE
FROM ACCOUNT
WHERE BALANCE = (SELECT MAX(BALANCE) FROM ACCOUNT);

-- d)
SELECT COUNT(*) AS TOTAL_CUSTOMERS
FROM CUSTOMER
WHERE ADDRESS IN ('Delhi','Mumbai');

-- e)
SELECT SUM(AMOUNT) AS TOTAL_LOAN
FROM LOAN
WHERE BRANCH_CODE IN ('B001','B002','B004');

-- f)
SELECT ACC_TYPE, COUNT(*) AS TOTAL_ACCOUNTS, SUM(BALANCE) AS TOTAL_BALANCE
FROM ACCOUNT
GROUP BY ACC_TYPE;

-- g)
SELECT TRANS_TYPE, SUM(TRANS_AMOUNT) AS TOTAL_AMOUNT
FROM TRANSACTION_DETAILS
WHERE ACC_NO <> 'A1005'
GROUP BY TRANS_TYPE;

-- h)
SELECT SUM(BALANCE) AS TOTAL_CURRENT_BALANCE
FROM ACCOUNT
WHERE ACC_TYPE = 'Current';

-- i)
SELECT ACC_TYPE, COUNT(*) AS NUM_ACCOUNTS
FROM ACCOUNT
GROUP BY ACC_TYPE
ORDER BY NUM_ACCOUNTS ASC;

-- j)
SELECT TRANS_TYPE, COUNT(*) AS NUM_TRANSACTIONS
FROM TRANSACTION_DETAILS
GROUP BY TRANS_TYPE
ORDER BY NUM_TRANSACTIONS DESC;

-- k)
SELECT ACC_TYPE, AVG(BALANCE) AS AVG_BALANCE
FROM ACCOUNT
GROUP BY ACC_TYPE
HAVING AVG(BALANCE) > 25000;

-- l)
SELECT ACC_TYPE, COUNT(*) AS NUM_ACCOUNTS
FROM ACCOUNT
GROUP BY ACC_TYPE
HAVING COUNT(*) > 2;

-- m)
SELECT ACC_TYPE, MAX(BALANCE) AS MAX_BALANCE
FROM ACCOUNT
GROUP BY ACC_TYPE
HAVING MAX(BALANCE) > 40000;

-- n)
SELECT MAX(BALANCE) AS HIGHEST_SAVINGS
FROM ACCOUNT
WHERE ACC_TYPE = 'Savings';

-- o) (Optional)
SELECT TRANS_TYPE
FROM TRANSACTION_DETAILS
GROUP BY TRANS_TYPE
HAVING SUM(TRANS_AMOUNT) = (
    SELECT MAX(SUM_AMOUNT)
    FROM (
        SELECT SUM(TRANS_AMOUNT) AS SUM_AMOUNT
        FROM TRANSACTION_DETAILS
        GROUP BY TRANS_TYPE
    )
);

-- p) (Optional)
SELECT ACC_TYPE
FROM ACCOUNT
GROUP BY ACC_TYPE
HAVING SUM(BALANCE) > (SELECT AVG(BALANCE) FROM ACCOUNT);
